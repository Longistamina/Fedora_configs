###########################
## 1. Install fish shell ##
###########################

# Fedora
sudo dnf install fish

# Ubuntu
sudo apt-add-repository ppa:fish-shell/release-4
sudo apt update
sudo apt install fish

#########################################################
## 2. Set fish shell as as actual system default shell ##
#########################################################

# 1. Run this command in Nushell to append your Homebrew Nushell path to a security file called /etc/shells (for fish shell, it's already there)
echo /usr/bin/fish | sudo tee -a /etc/shells

# 2. Set Nushell as your default shell
chsh -s /usr/bin/fish

# 3. If you set "Run a custom command instead of my shell", untick it

# 4. Reboot/Restart computer to apply changes
