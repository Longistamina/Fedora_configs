#############################
## Install and launch niri ##
#############################

sudo dnf install niri

Open "Alarchy" terminal
=> run "niri-session"

#######################
## Vietnamese typing ##
#######################

sudo dnf install fcitx5 fcitx5-bamboo fcitx5-configtool

Open "~/.config/niri/config.kdl"
=> add 'spawn-at-startup "fcitx5 -d --replace"'
=> add
environment {
    GTK_IM_MODULE "fcitx"
    QT_IM_MODULE "fcitx"
    XMODIFIERS "@im=fcitx"
    GLFW_IM_MODULE "ibus" // Some apps, like Minecraft, use this variable
}

git clone https://github.com/fcitx/fcitx5-bamboo.git
cd fcitx5-bamboo

git submodule init
git submodule update

mkdir build
cd build
cmake -DCMAKE_INSTALL_PREFIX=/usr ..
make
sudo make install

# Set Alt+Z to switch input
fcitx5-configtool -> Global Options -> Toggle Input Method -> ALt+Z

################################
