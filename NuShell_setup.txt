#####################################
## (OPTIONAL) Install via homebrew ##
#####################################

# Follow the instruction of this page to install homebrew: https://brew.sh/
/bin/bash -c "$(curl -fsSL https://raw.githubusercontent.com/Homebrew/install/HEAD/install.sh)"

# Use homebrew to install Nushell
brew install nushell

########################
## Install for ubuntu ##
########################

wget -qO- https://apt.fury.io/nushell/gpg.key | sudo gpg --dearmor -o /etc/apt/keyrings/fury-nushell.gpg
echo "deb [signed-by=/etc/apt/keyrings/fury-nushell.gpg] https://apt.fury.io/nushell/ /" | sudo tee /etc/apt/sources.list.d/fury-nushell.list
sudo apt update
sudo apt install nushell

# 3. Right click
# 4. Choose "Preferences"
# 5. Move to "Command"
# 6. Paste the path to "Run a custom command instead of my shell"

#########################
## Set up "config nu " ##
#########################

# 1. Open $nu.config-path using nano or vscode
code $nu.config-path
nano $nu.config-path

# 2. Set the config.buffer_editor (at this to the bottom of the file)
$env.config.buffer_editor = "gnome-text-editor" # (Can also be "code", "vim", ...)

# 3. Add this line in the file to in order to disable the welcome message
$env.config.show_banner = false

# 4. Save, exit the file

# 5. Restart nu, then run "config nu" to automatically open the nu config file

######################
## Set up for Conda ##
######################

# 1. Inside NuShell, cd to ~, then run this
git clone https://github.com/nushell/nu_scripts.git ~/.nu_scripts

# 2. Open config file
config nu

# 3. Once the config.nu file is open, 
# scroll all the way to the bottom and paste the module line from the previous step
use ~/.nu_scripts/modules/virtual_environments/conda.nu *

# 4. Set nushell as default profile in vscode, add this to settings.json (Preferences: Open User Settings (JSON))
"terminal.integrated.profiles.linux": {
        "Nushell": {
            "path": "nu",
            "icon": "terminal-bash"
        }
    },
    "terminal.integrated.defaultProfile.linux": "Nushell"

# 5. Activate/Deactivate an env
activate env_name # (NOTE: not conda activate env_name)
deactivate

# 6. Set env.path
code $nu.env-path # (or nano $nu.env-path)
$env.PATH = ($env.PATH | split row (char esep) | prepend '/home/linuxbrew/.linuxbrew/bin' | prepend '/home/longdpt/miniconda3/bin' | prepend '/home/longdpt/.local/bin' | uniq) # add to the bottom
activate env_name # (Optionally, can add this to the bottom after the above line to make nu automatically activate env)

# 7. Set some "terminal.activate..." to false (Preferences: Open User Settings (JSON))
"python.terminal.activateEnvironment": false,
"python.terminal.activateEnvInCurrentTerminal": false,

######################################
## Alternative of "source ~/bashrc" ##
######################################

In nu, instead of using , use this:
source $nu.config-path # Apply new config change
source $nu.env-path # Apply new path change

##################################
## Oh-My-Posh setup for nushell ##
##################################

# 1. First, locate the path/to/theme.omp.json, for example:
/home/longdpt/.config/oh-my-posh/quick_term.omp.json

# 2. Run this command
oh-my-posh init nu --config /home/longdpt/.config/oh-my-posh/quick_term.omp.json | save -f ~/.oh-my-posh.nu

# 3. Open config file
config nu
code $nu.config-path

# 4. Add this to the end of the file
source ~/.oh-my-posh.nu

# 5. Run this to update changes
source $nu.config-path # Apply new config change
